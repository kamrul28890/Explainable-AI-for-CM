# Shared dependencies for every platform.
#
# torch is NOT listed here. It must be installed FIRST, from a platform-specific
# index, because installing it as a transitive dependency silently pulls a
# CPU-only wheel and every CUDA run then falls back to the CPU without warning.
# See docs/setup/SETUP.md for the exact torch command per machine.

# --- core model stack ------------------------------------------------------
# transformers is pinned to 4.49.0 for two independent reasons:
#   1. >= 5.0 removed legacy generation attributes that Florence-2's remote code
#      (configuration_florence2.py) needs, crashing the pilot arm on load.
#   2. 4.49.0 is the first release with native Qwen2.5-VL support.
# It is the only version where BOTH studies run in one environment.
transformers==4.49.0
huggingface_hub==0.36.2
tokenizers==0.21.4
datasets==4.5.0
accelerate>=1.0

# 4-bit / 8-bit quantisation. CUDA-only -- there is no Apple Silicon build, which
# is why macOS uses MLX weights instead (see requirements/mac.txt).
bitsandbytes>=0.43; platform_system != "Darwin"

# --- data and numerics -----------------------------------------------------
# numpy 2.x is what the verified environment runs (2.2.6). Do not pin below 2:
# it would downgrade a working install and break compiled wheels built against 2.x.
numpy>=2.0
pandas>=2.0
pyarrow>=14.0
scipy>=1.11

# --- imaging ---------------------------------------------------------------
Pillow>=10.0
opencv-python>=4.8
scikit-image>=0.22

# --- explanation-correctness grading (metric 7) ----------------------------
# Small (~90 MB) sentence-embedding model for semantic similarity between the
# model's rationale and the human-written reason.
sentence-transformers>=3.0

# --- figures and notebooks -------------------------------------------------
matplotlib>=3.8
seaborn>=0.13
jupyter
# ipykernel 7.x rewrote the shell channel as async and added the subshell
# protocol; VS Code's Jupyter extension (2025.9.1) hangs forever on
# "Connecting to kernel..." against it, while nbconvert works fine -- which makes
# it look like a notebook bug rather than a client mismatch.
ipykernel<7

# --- model-specific helpers ------------------------------------------------
einops
timm

# --- testing ---------------------------------------------------------------
pytest>=8.0
